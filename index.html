            container.appendChild(renderer.domElement);


            // Create Geometry (Sphere made of points)

            // Increased segments for higher dot density

            const geometry = new THREE.SphereGeometry(14, 80, 80);

                opacity: 0.8,

                sizeAttenuation: true

            });


            // Combine into Points Mesh

            const sphere = new THREE.Points(geometry, material);

            

            // Tilt the sphere slightly on the X-axis for a more dynamic structural look

            sphere.rotation.x = Math.PI / 8;

            scene.add(sphere);


            // Add subtle fog to fade out the back/bottom points of the sphere smoothly

            scene.fog = new THREE.FogExp2(0x0a0a0a, 0.025);

            function animate() {
